  <div class="modal">

    <label>Количество пачек:</label>

    <input type="number" id="modalQty" min="0" step="1">

    <label>Дата последней поставки:</label>

    <div class="modal-buttons">

<script>

 const INGREDIENTS = [

    { key: 'coffee',    name: '☕ Кофе',                       unit: 'пачек по 1 кг' },

    { key: 'milk',      name: '🥛 Молоко',                     unit: 'пачек по 1 кг' },

    { key: 'chocolate', name: '🍫 Горячий шоколад',            unit: 'пачек по 1 кг' },

    { key: 'raf',       name: '🍮 Раф',                        unit: 'пачек по 1 кг' },

    { key: 'sugar',     name: '🍬 Сахар',                      unit: 'коробок' },

    { key: 'cups',      name: '🥤 Стаканы',                    unit: 'коробок' },

    { key: 'lids',      name: '🧢 Крышки',                     unit: 'коробок' },

    { key: 'sticks',    name: '🥄 Палочки для размешивания',   unit: 'упаковок' },

    { key: 'water',     name: '💧 Вода',                       unit: 'бутылей 19 л' }

  ];


  let points = JSON.parse(localStorage.getItem('vendingPoints') || '[]');

  let editing = { pointId: null, ingKey: null };


  function save() {

    localStorage.setItem('vendingPoints', JSON.stringify(points));

  }


  function addPoint() {

    const input = document.getElementById('newPointName');

    const name = input.value.trim();

    if (!name) {

      alert('Введите название точки');

      return;

    }

    const newPoint = {

      id: Date.now(),

      name: name,

      ingredients: {}

    };

    INGREDIENTS.forEach(ing => {

      newPoint.ingredients[ing.key] = { qty: 0, date: '' };

    });

    points.push(newPoint);

    input.value = '';

    save();

    render();

  }


  function deletePoint(id) {

    if (confirm('Удалить точку?')) {

      points = points.filter(p => p.id !== id);

      save();

      render();

    }

  }


  function openModal(pointId, ingKey) {

    editing = { pointId, ingKey };

    const point = points.find(p => p.id === pointId);

    const ing = INGREDIENTS.find(i => i.key === ingKey);

    document.getElementById('modalTitle').textContent = ing.name;

    document.getElementById('modalQty').value = point.ingredients[ingKey].qty || 0;

    document.getElementById('modalDate').value = point.ingredients[ingKey].date || new Date().toISOString().slice(0,10);

    document.getElementById('editModal').classList.add('show');

  }


  function closeModal() {

    document.getElementById('editModal').classList.remove('show');

  }


  function saveEdit() {

    const point = points.find(p => p.id === editing.pointId);

    const qty = parseInt(document.getElementById('modalQty').value) || 0;

    const date = document.getElementById('modalDate').value;

    point.ingredients[editing.ingKey] = { qty, date };

    save();

    closeModal();

    render();

  }


  function getDaysSince(dateStr) {

    if (!dateStr) return null;

    const today = new Date();

    today.setHours(0,0,0,0);

    const d = new Date(dateStr);

    d.setHours(0,0,0,0);


return Math.floor((today - d) / (1000*60*60*24));

  }


  function getStatus(days) {

    if (days === null) return { cls: '', label: 'нет данных', tag: '' };

    if (days < 3) return { cls: 'ok', label: 'нормально', tag: 'status-ok' };

    if (days <= 7) return { cls: 'soon', label: 'скоро подвозить', tag: 'status-soon' };

    return { cls: 'urgent', label: 'срочно!', tag: 'status-urgent' };

  }


  function render() {

    const container = document.getElementById('pointsList');

    if (points.length === 0) {

      container.innerHTML = `

        <div class="empty-state">

          <span>📍</span>

          <p>Нет добавленных точек.<br>Добавьте первую точку выше ☝️</p>

        </div>`;

      return;

    }


    container.innerHTML = points.map(point => `

      <div class="point-card">

        <div class="point-header">

          <div class="point-name">📍 ${point.name}</div>

          <button class="btn-danger btn-small" onclick="deletePoint(${point.id})">Удалить</button>

        </div>

        ${INGREDIENTS.map(ing => {

          const data = point.ingredients[ing.key] || { qty: 0, date: '' };

          const days = getDaysSince(data.date);

          const status = getStatus(days);

          const dateStr = data.date ? new Date(data.date).toLocaleDateString('ru-RU') : '—';

          const daysStr = days !== null ? `${days} дн. назад` : 'не указано';

          return `

            <div class="ingredient ${status.cls}">

              <div class="ing-info">

                <div class="ing-name">${ing.name}</div>

                <div class="ing-details">

                  ${data.qty} ${ing.unit} • поставка: ${dateStr} (${daysStr})

                </div>

                ${data.date ? `<span class="ing-status ${status.tag}">${status.label}</span>` : ''}

              </div>

              <button class="edit-btn" onclick="openModal(${point.id}, '${ing.key}')">✏️</button>

            </div>

          `;

        }).join('')}

      </div>

    `).join('');

  }


  function resetAll() {

    if (confirm('⚠️ Точно сбросить ВСЕ данные? Это действие нельзя отменить!')) {

      if (confirm('Вы уверены? Все точки и поставки будут удалены.')) {

        points = [];

        localStorage.removeItem('vendingPoints');

        render();

      }

    }

  }


  // Закрытие модалки по клику вне окна

  document.getElementById('editModal').addEventListener('click', (e) => {

    if (e.target.id === 'editModal') closeModal();

  });


  render();

</script>
